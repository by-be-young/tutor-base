   184	const authStore = useAuthStore()
   185	const checkinStore = useCheckinStore()
   186	const tasksStore = useTasksStore()
   187	
   188	// ========== 常量 ==========
   189	const TASK_POINTS = 100
   190	const MAX_DISPLAY = 6000 // 进度条展示至 6000，超出当前积分的里程碑用于演示「未达到条件」
   191	
   192	// 任务列表（仅保留每日签到；签到状态与积分由后端 /api/v1/checkins 提供）
   193	const mockTasks = [
   194	    { id: 4, name: '每日签到', desc: '登录并完成当日签到', icon: 'fas fa-calendar-check', color: '#7d5ca8' }
   195	]
   196	
   197	// ========== 里程碑 ==========
   198	// 生成 200 ~ 6000 的里程碑；每 1000 积分为稀有卡片节点，其余为普通卡片节点
   199	const milestones = buildMilestones(MAX_DISPLAY)
   200	const reachedCount = computed(() => milestones.filter(m => m.pts <= tasksStore.points).length)
   201	
   202	// ========== 里程碑状态 ==========
   203	function milestoneClass(m) {
   204	    if (m.pts > tasksStore.points) return 'is-locked'
   205	    if (tasksStore.claimedMilestones.has(m.pts)) return 'is-claimed'
   206	    return 'is-claimable'
   207	}
   208	
   209	function milestoneIcon(m) {
   210	    if (m.pts > tasksStore.points) return 'fas fa-lock'
   211	    if (tasksStore.claimedMilestones.has(m.pts)) return 'fas fa-check'
   212	    return m.isRare ? 'fas fa-star' : 'fas fa-gift'
   213	}
   214	
   215	function milestoneLabel(m) {
   216	    if (m.pts > tasksStore.points) return '未达到'
   217	    if (tasksStore.claimedMilestones.has(m.pts)) return '已领取'
   218	    return m.isRare ? '稀有·待领取' : '待领取'
   219	}
   220	
   221	function connectorClass(m) {
   222	    return m.pts <= tasksStore.points ? 'is-reached' : 'is-locked'
   223	}
   224	
   225	// ========== 进度条滚动 ==========
   226	// 默认/领取后停靠右侧：已领取的部分被隐藏到左侧，可通过「查看已领取」或手动滚动查看
   227	const trackScroll = ref(null)
   228	
   229	function scrollToClaimed() {
   230	    trackScroll.value?.scrollTo({ left: 0, behavior: 'smooth' })
   231	}
   232	
   233	function scrollToLatest() {
   234	    nextTick(() => {
   235	        trackScroll.value?.scrollTo({ left: trackScroll.value.scrollWidth, behavior: 'smooth' })
   236	    })
   237	}
   238	
   239	// ========== 领取卡片 ==========
   240	const claimVisible = ref(false)
   241	const flipped = ref(false)
   242	const currentReward = ref(null)
   243	const claimingMilestone = ref(false)
   244	let flipTimer = null
   245	
   246	function openClaim(m) {
   247	    if (m.pts > tasksStore.points || tasksStore.claimedMilestones.has(m.pts)) return
   248	    // 里程碑 → 卡片由目录确定性映射（同一里程碑所有人获得同一张卡）
   249	    currentReward.value = { milestone: m, card: m.card }
   250	    flipped.value = false
   251	    claimVisible.value = true
   252	    // 先展示背面，再翻转揭晓
   253	    flipTimer = setTimeout(() => {
   254	        flipped.value = true
   255	    }, 900)
   256	}
   257	
   258	async function confirmClaim() {
   259	    if (!currentReward.value || !flipped.value || claimingMilestone.value) return
   260	    claimingMilestone.value = true
   261	    try {
   262	        const card = await tasksStore.claimMilestone(authStore.currentUser, currentReward.value.milestone)
   263	        claimVisible.value = false
   264	        showToast(`卡片「${card.name}」已收入收藏室`, 'success')
   265	        scrollToLatest()
   266	    } catch (e) {
   267	        showToast(e.message, 'error')
   268	    } finally {
   269	        claimingMilestone.value = false
   270	    }
   271	}
   272	
   273	function closeClaim() {
   274	    if (!flipped.value || claimingMilestone.value) return // 揭晓动画进行中不允许关闭
   275	    claimVisible.value = false
   276	}
   277	
   278	function cardGradient(card) {
   279	    if (!card) return 'linear-gradient(160deg, #5ba8a4, #3a7d79)'
   280	    return `linear-gradient(160deg, ${card.colors[0]}, ${card.colors[1]})`

   298	// ========== Toast（与错题本保持一致） ==========
   299	let toastTimer = null
   300	function showToast(message, type = 'info', duration = 3000) {
   301	    const existing = document.querySelector('.custom-toast')
   302	    if (existing) {
   303	        existing.remove()
   304	        if (toastTimer) {
   305	            clearTimeout(toastTimer)
   306	            toastTimer = null
   307	        }
   308	    }
   309	
   310	    const toast = document.createElement('div')
   311	    toast.className = `custom-toast toast-${type}`
   312	    toast.textContent = message
   313	    document.body.appendChild(toast)
   314	
   315	    requestAnimationFrame(() => toast.classList.add('toast-visible'))
   316	
   317	    toastTimer = setTimeout(() => {
   318	        toast.classList.remove('toast-visible')
   319	        setTimeout(() => {
   320	            if (toast.parentNode) toast.parentNode.removeChild(toast)
   321	            toastTimer = null
   322	        }, 300)
   323	    }, duration)
   324	}
